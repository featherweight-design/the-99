# The 99 v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship v1 of The 99: upload a ManaBox collection CSV, get ranked explainable commander recommendations, and build a Commander deck from owned cards with unowned gaps flagged.

**Architecture:** Single Next.js (App Router) app on Vercel with Neon Postgres via Drizzle ORM and Clerk auth. A Scryfall bulk-data seed job populates a local `cards` catalog; recommendation scoring runs as pure TypeScript functions over DB queries; EDHREC synergy data is fetched per-commander and cached in Postgres.

**Tech Stack:** Next.js 15 (App Router, TypeScript), Drizzle ORM + `@neondatabase/serverless`, Clerk, Vitest, Tailwind CSS, Zod.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-07-18-the-99-design.md` — implement exactly; v1 scope only (no deck import, no win/loss tracking, no LLM).
- All user-scoped tables carry Clerk `user_id` (text). Every server action/route must scope queries by the authenticated user.
- Pure-logic modules (parsing, bucketing, scoring, slugs, export) must have no DB or network imports — testable with fixtures only; tests never hit the network. DB/network-aware modules (`seed.ts`, `edhrec/client.ts`, `recs/recommend.ts`, `auth.ts`) also live under `src/lib/**` but keep their pure parts split into sibling modules.
- CSV re-upload fully replaces the user's collection (transactional delete + insert), only after user confirmation.
- EDHREC is unofficial: 30-day cache TTL, serve stale on failure, degrade to Scryfall-only scoring with a visible notice when no data exists.
- Role targets: ramp 10+, draw 10+, removal 8+.
- Commit style: conventional commits (`feat:`, `test:`, `chore:`, `docs:`).
- Merge policy: squash-merge PRs into `main`; Copilot review per user-level policy before merge.
- UI styling: neutral Tailwind for now — a Claude Design brand pass (palette/type/icon) lands later; do not invent a strong visual identity.

```ts
export type ManaBoxRow = {
  name: string; setCode: string; quantity: number;
  scryfallId: string | null; foil: boolean; condition: string | null;
};
export type ParseError = { line: number; raw: string; reason: string };

function splitCsvLine(line: string): string[] {
  const out: string[] = [];
  let cur = "", inQuotes = false;
  for (let i = 0; i < line.length; i++) {
    const ch = line[i];
    if (inQuotes) {
      if (ch === '"' && line[i + 1] === '"') { cur += '"'; i++; }
      else if (ch === '"') inQuotes = false;
      else cur += ch;
    } else if (ch === '"') inQuotes = true;
    else if (ch === ",") { out.push(cur); cur = ""; }
    else cur += ch;
  }
  out.push(cur);
  return out;
}

const REQUIRED = ["Name", "Set code", "Foil", "Quantity", "Scryfall ID", "Condition"] as const;

export function parseManaBoxCsv(csvText: string): { rows: ManaBoxRow[]; errors: ParseError[] } {
  // Keep original indices (blank lines skipped in the loop) so error line numbers match the file.
  const lines = csvText.split(/\r?\n/);
  if (lines.every((l) => l.trim() === "")) return { rows: [], errors: [{ line: 1, raw: "", reason: "empty file" }] };

  const header = splitCsvLine(lines[0]);
  const idx = Object.fromEntries(REQUIRED.map((c) => [c, header.indexOf(c)]));
  const missing = REQUIRED.filter((c) => idx[c] === -1);
  if (missing.length > 0) {
    return { rows: [], errors: [{ line: 1, raw: lines[0], reason: `header missing columns: ${missing.join(", ")}` }] };
  }

  const rows: ManaBoxRow[] = [];
  const errors: ParseError[] = [];
  for (let i = 1; i < lines.length; i++) {
    if (lines[i].trim() === "") continue;
    const cols = splitCsvLine(lines[i]);
    if (cols.length < header.length) {
      errors.push({ line: i + 1, raw: lines[i], reason: `expected ${header.length} columns, got ${cols.length}` });
      continue;
    }
    const quantity = Number(cols[idx["Quantity"]]);
    if (!Number.isInteger(quantity) || quantity < 1) {
      errors.push({ line: i + 1, raw: lines[i], reason: `invalid quantity "${cols[idx["Quantity"]]}"` });
      continue;
    }
    rows.push({
      name: cols[idx["Name"]].trim(),
      setCode: cols[idx["Set code"]].trim(),
      quantity,
      scryfallId: cols[idx["Scryfall ID"]]?.trim() || null,
      foil: cols[idx["Foil"]].trim().toLowerCase() !== "normal",
      condition: cols[idx["Condition"]]?.trim() || null,
    });
  }
  return { rows, errors };
}
```

```ts
export type Role = "ramp" | "draw" | "removal" | "land";
export type RoleInput = { typeLine: string; oracleText: string | null; keywords: string[] };

export const ROLE_TARGETS: Record<Exclude<Role, "land">, number> = {
  ramp: 10, draw: 10, removal: 8,
};

const RAMP_PATTERNS = [
  /\{t\}: add /, /adds? (one|two|three|\{[wubrgc]\})/, 
  /search your library for .{0,40}land .{0,60}onto the battlefield/,
];
const DRAW_PATTERNS = [/draw (a|one|two|three|four|x) cards?/];
const REMOVAL_PATTERNS = [
  /destroy target/, /exile target/, /destroy all/, /exile all/,
  /counter target/, /deals? \d+ damage to (any target|target creature)/,
  /target creature gets [-−]\d/,
];
const NOT_REMOVAL = /target .{0,30}you control/;

export function classifyRoles(card: RoleInput): Role[] {
  const roles: Role[] = [];
  const text = (card.oracleText ?? "").toLowerCase();
  const type = card.typeLine.toLowerCase();

  if (type.includes("land")) roles.push("land");
  if (!type.includes("land") && RAMP_PATTERNS.some((p) => p.test(text))) roles.push("ramp");
  if (DRAW_PATTERNS.some((p) => p.test(text))) roles.push("draw");
  if (REMOVAL_PATTERNS.some((p) => p.test(text)) && !NOT_REMOVAL.test(text)) roles.push("removal");
  return roles;
}
```

```ts
// Explicit shape (kept in sync with the cards table) rather than a schema
// import: pure lib code stays free of DB imports per the global constraints.
export type CardRow = {
  oracleId: string; scryfallId: string; name: string; manaCost: string | null;
  cmc: number; colorIdentity: string[]; typeLine: string; oracleText: string | null;
  keywords: string[]; commanderLegal: boolean; imageUri: string | null; rarity: string;
};
export type ScryfallCard = Record<string, any>;

const SKIP_LAYOUTS = new Set(["token", "double_faced_token", "emblem", "art_series"]);

export function toCardRow(raw: ScryfallCard): CardRow | null {
  if (!raw.oracle_id || SKIP_LAYOUTS.has(raw.layout)) return null;
  const faces = raw.card_faces as ScryfallCard[] | undefined;
  const oracleText =
    raw.oracle_text ??
    (faces ? faces.map((f) => f.oracle_text ?? "").join("\n//\n") : null);
  const imageUri =
    raw.image_uris?.normal ?? faces?.find((f) => f.image_uris?.normal)?.image_uris.normal ?? null;
  return {
    oracleId: raw.oracle_id,
```

```ts
import { Readable } from "node:stream";
import { sql } from "drizzle-orm";
import { chain } from "stream-chain";
import { parser } from "stream-json";
import { streamArray } from "stream-json/streamers/StreamArray";
import { db } from "../../db";
import { cards } from "../../db/schema";
import { toCardRow, type CardRow } from "./transform";

const BULK_INDEX = "https://api.scryfall.com/bulk-data/oracle-cards";

// The bulk file is a ~150MB JSON array; stream-parse it so the seed never
// materializes the whole payload (keeps the Vercel cron route within memory limits).
export async function seedCards(): Promise<{ upserted: number; skipped: number }> {
  const meta = await fetch(BULK_INDEX, { headers: { "User-Agent": "the-99/1.0" } }).then((r) => {
    if (!r.ok) throw new Error(`bulk-data index ${r.status}`);
    return r.json();
  });
  const res = await fetch(meta.download_uri, { headers: { "User-Agent": "the-99/1.0" } });
  if (!res.ok || !res.body) throw new Error(`bulk download ${res.status}`);

  let upserted = 0, skipped = 0;
  let batch: CardRow[] = [];
  const flush = async () => {
    if (batch.length === 0) return;
    await db.insert(cards).values(batch).onConflictDoUpdate({
      target: cards.oracleId,
      // Full-column update via Postgres excluded refs keeps the upsert idempotent.
      set: {
        scryfallId: sql`excluded.scryfall_id`,
        name: sql`excluded.name`,
        manaCost: sql`excluded.mana_cost`,
        cmc: sql`excluded.cmc`,
        colorIdentity: sql`excluded.color_identity`,
        typeLine: sql`excluded.type_line`,
        oracleText: sql`excluded.oracle_text`,
        keywords: sql`excluded.keywords`,
        commanderLegal: sql`excluded.commander_legal`,
        imageUri: sql`excluded.image_uri`,
        rarity: sql`excluded.rarity`,
      },
    });
    upserted += batch.length;
    batch = [];
  };

  const pipeline = chain([
    Readable.fromWeb(res.body as import("node:stream/web").ReadableStream),
    parser(),
    streamArray(),
  ]);
  for await (const { value } of pipeline) {
    const row = toCardRow(value);
    if (!row) { skipped++; continue; }
    batch.push(row);
    if (batch.length >= 500) await flush();
  }
  await flush();
  return { upserted, skipped };
}
```

```ts
import type { ManaBoxRow } from "../manabox/parse";

export type CatalogCard = { oracleId: string; scryfallId: string; name: string };
export type MatchResult = {
  matched: { row: ManaBoxRow; oracleId: string }[];
  unmatched: ManaBoxRow[];
};

export function matchRows(rows: ManaBoxRow[], catalog: CatalogCard[]): MatchResult {
  // Name matching is the primary path for v1: the catalog is seeded from
  // Scryfall oracle-cards (one arbitrary printing per oracle id), while
  // ManaBox "Scryfall ID" values are printing-level ids — the id map only
  // hits when the collection happens to contain that exact printing.
  const byScryfallId = new Map(catalog.map((c) => [c.scryfallId, c.oracleId]));
  const byName = new Map<string, string>();
  for (const c of catalog) {
    const lower = c.name.toLowerCase();
    byName.set(lower, c.oracleId);
    const front = lower.split(" // ")[0];
    if (!byName.has(front)) byName.set(front, c.oracleId);
  }

  const matched: MatchResult["matched"] = [];
  const unmatched: ManaBoxRow[] = [];
  for (const row of rows) {
    const oracleId =
      (row.scryfallId && byScryfallId.get(row.scryfallId)) ??
      byName.get(row.name.toLowerCase());
    if (oracleId) matched.push({ row, oracleId });
    else unmatched.push(row);
  }
  return { matched, unmatched };
}
```

```ts
"use server";

import { and, eq, sql } from "drizzle-orm";
import { revalidatePath } from "next/cache";
import { db } from "../../db";
import { cards, collectionItems, csvUploads } from "../../db/schema";
import { requireUserId } from "../../lib/auth";
import { parseManaBoxCsv } from "../../lib/manabox/parse";
import { matchRows } from "../../lib/collection/match";

export type UploadPreview = {
  uploadId: string;
  matchedCardCount: number; // total card quantity across matched rows
  unmatchedRows: { name: string; setCode: string }[];
  parseErrors: { line: number; reason: string }[];
  currentCollectionCount: number; // total card quantity in the current collection
};

type PendingItem = { oracleId: string; setCode: string; foil: boolean; condition: string | null; quantity: number };

// Matched rows are aggregated to compact (oracleId,setCode,foil) totals before
// being stashed on the upload record: the JSONB stays small for large collections
// and confirm becomes a plain insert.
export async function previewUpload(formData: FormData): Promise<UploadPreview> {
  const userId = await requireUserId();
  const file = formData.get("file");
  if (!(file instanceof File)) throw new Error("No CSV file provided");
  const { rows, errors } = parseManaBoxCsv(await file.text());

  const catalog = await db
    .select({ oracleId: cards.oracleId, scryfallId: cards.scryfallId, name: cards.name })
    .from(cards);
  const { matched, unmatched } = matchRows(rows, catalog);

  const agg = new Map<string, PendingItem>();
  for (const { row, oracleId } of matched) {
    const key = `${oracleId}|${row.setCode}|${row.foil}`;
    const existing = agg.get(key);
    if (existing) {
      existing.quantity += row.quantity;
      // Same printing in multiple conditions collapses to one row (PK excludes condition);
      // a mixed-condition stack has no single truthful condition, so store null.
      if (existing.condition !== row.condition) existing.condition = null;
    } else {
      agg.set(key, { oracleId, setCode: row.setCode, foil: row.foil, condition: row.condition, quantity: row.quantity });
    }
  }
  const pendingItems = [...agg.values()];
  const matchedCardCount = pendingItems.reduce((n, i) => n + i.quantity, 0);

  const [{ count: currentCollectionCount }] = await db
    .select({ count: sql<number>`coalesce(sum(${collectionItems.quantity}), 0)::int` })
    .from(collectionItems)
    .where(eq(collectionItems.userId, userId));

  const [upload] = await db.insert(csvUploads).values({
    userId,
    filename: file.name,
    rowCount: rows.length,
    unmatchedCount: unmatched.length,
    previewPayload: { unmatched, parseErrors: errors, pendingItems },
  }).returning({ id: csvUploads.id });

  return {
    uploadId: upload.id,
    matchedCardCount,
    unmatchedRows: unmatched.map((r) => ({ name: r.name, setCode: r.setCode })),
    parseErrors: errors.map((e) => ({ line: e.line, reason: e.reason })),
    currentCollectionCount,
  };
}

export async function confirmUpload(uploadId: string): Promise<void> {
  const userId = await requireUserId();
  const [upload] = await db.select().from(csvUploads)
    .where(and(eq(csvUploads.id, uploadId), eq(csvUploads.userId, userId)));
  if (!upload) throw new Error("upload not found");

  const { pendingItems } = upload.previewPayload as { pendingItems: PendingItem[] };
  const values = pendingItems.map((v) => ({ userId, ...v }));
  // Full replace, transactional.
  await db.transaction(async (tx) => {
    await tx.delete(collectionItems).where(eq(collectionItems.userId, userId));
    for (let i = 0; i < values.length; i += 500) {
      await tx.insert(collectionItems).values(values.slice(i, i + 500));
    }
  });
  revalidatePath("/collection");
}
```

```ts
  const cardlists = (payload as any)?.container?.json_dict?.cardlists;
```

```ts
import { eq } from "drizzle-orm";
import { db } from "../../db";
import { edhrecCache } from "../../db/schema";
import { commanderSlug } from "./slug";
import { parseEdhrecPayload, type EdhrecData } from "./parse";

const TTL_MS = 30 * 24 * 60 * 60 * 1000;

export async function getEdhrecData(
  commanderOracleId: string,
  commanderName: string,
): Promise<EdhrecData | null> {
  const [cached] = await db.select().from(edhrecCache)
    .where(eq(edhrecCache.commanderOracleId, commanderOracleId));
  const fresh = cached && Date.now() - cached.fetchedAt.getTime() < TTL_MS;
  if (fresh) return parseEdhrecPayload(cached.payload);

  try {
    const res = await fetch(
      `https://json.edhrec.com/pages/commanders/${commanderSlug(commanderName)}.json`,
      // Timeout so a hung EDHREC request falls through to stale/degraded
      // handling instead of stalling the recommendations page.
      { headers: { "User-Agent": "the-99/1.0" }, signal: AbortSignal.timeout(10_000) },
    );
    if (!res.ok) throw new Error(`edhrec ${res.status}`);
    const payload = await res.json();
    await db.insert(edhrecCache)
      .values({ commanderOracleId, payload, fetchedAt: new Date() })
      .onConflictDoUpdate({
        target: edhrecCache.commanderOracleId,
        set: { payload, fetchedAt: new Date() },
      });
    return parseEdhrecPayload(payload);
  } catch {
    if (cached) return parseEdhrecPayload(cached.payload); // stale-on-error
    return null;
  }
}
```

```ts
import { classifyRoles, ROLE_TARGETS } from "./roles";

export type PoolCard = {
  oracleId: string; name: string; typeLine: string; oracleText: string | null;
  keywords: string[]; colorIdentity: string[];
};
export type CommanderScore = {
  oracleId: string; name: string;
  colorPoolSize: number;
  edhrecOwnedCount: number | null;
  edhrecTotal: number | null;
  roleCounts: Record<"ramp" | "draw" | "removal", number>;
  roleGaps: ("ramp" | "draw" | "removal")[];
  score: number;
};

export function isCommanderCandidate(card: Pick<PoolCard, "typeLine" | "oracleText">): boolean {
  if (/legendary/i.test(card.typeLine) && /creature/i.test(card.typeLine)) return true;
  return /can be your commander/i.test(card.oracleText ?? "");
}

function fitsIdentity(card: PoolCard, identity: Set<string>): boolean {
  return card.colorIdentity.every((c) => identity.has(c));
}

export function scoreCommander(args: {
  commander: PoolCard;
  collection: PoolCard[];
  edhrecTopCards: string[] | null;
}): CommanderScore {
  const identity = new Set(args.commander.colorIdentity);
  const pool = args.collection.filter(
    (c) => c.oracleId !== args.commander.oracleId && fitsIdentity(c, identity),
  );

  const roleCounts = { ramp: 0, draw: 0, removal: 0 };
  for (const c of pool) {
    for (const role of classifyRoles(c)) {
      if (role !== "land") roleCounts[role]++;
    }
  }
  const roleGaps = (Object.keys(ROLE_TARGETS) as (keyof typeof ROLE_TARGETS)[])
    .filter((r) => roleCounts[r] < ROLE_TARGETS[r]);

  let edhrecOwnedCount: number | null = null;
  let edhrecTotal: number | null = null;
  if (args.edhrecTopCards) {
    const poolNames = new Set(pool.map((c) => c.name.toLowerCase()));
    edhrecTotal = args.edhrecTopCards.length;
    edhrecOwnedCount = args.edhrecTopCards
      .filter((n) => poolNames.has(n.toLowerCase())).length;
  }

  // EDHREC overlap dominates; pool size and role coverage round it out.
  const overlapRatio = edhrecTotal ? (edhrecOwnedCount! / edhrecTotal) : 0;
  const roleCoverage =
    (Object.keys(ROLE_TARGETS) as (keyof typeof ROLE_TARGETS)[])
      .reduce((sum, r) => sum + Math.min(roleCounts[r], ROLE_TARGETS[r]) / ROLE_TARGETS[r], 0) /
    Object.keys(ROLE_TARGETS).length;
  const poolFactor = Math.min(args.collection.length ? pool.length / 200 : 0, 1);
  const score = Math.round((overlapRatio * 60 + roleCoverage * 25 + poolFactor * 15) * 10) / 10;

  return {
    oracleId: args.commander.oracleId,
    name: args.commander.name,
    colorPoolSize: pool.length,
    edhrecOwnedCount, edhrecTotal,
    roleCounts, roleGaps, score,
  };
}
```

```ts
import { isCommanderCandidate, scoreCommander, type CommanderScore, type PoolCard } from "./score";

export async function recommendCommanders(
  userId: string,
): Promise<{ results: CommanderScore[]; degraded: boolean }> {
  const owned = await db
    .selectDistinctOn([cards.oracleId], {
      oracleId: cards.oracleId, name: cards.name, typeLine: cards.typeLine,
      oracleText: cards.oracleText, keywords: cards.keywords,
      colorIdentity: cards.colorIdentity, commanderLegal: cards.commanderLegal,
    })
    .from(collectionItems)
    .innerJoin(cards, eq(collectionItems.oracleId, cards.oracleId))
    .where(eq(collectionItems.userId, userId));

  const collection: PoolCard[] = owned;
  const candidates = owned.filter((c) => c.commanderLegal && isCommanderCandidate(c));

  // Fetch EDHREC data with bounded concurrency: sequential awaits would make
  // cold loads with dozens of candidates exceed the page's time budget.
  const CONCURRENCY = 5;
  let degraded = false;
  const results: CommanderScore[] = [];
  for (let i = 0; i < candidates.length; i += CONCURRENCY) {
    const scored = await Promise.all(
      candidates.slice(i, i + CONCURRENCY).map(async (commander) => {
        const edhrec = await getEdhrecData(commander.oracleId, commander.name);
        if (edhrec === null) degraded = true;
        return scoreCommander({ commander, collection, edhrecTopCards: edhrec?.topCards ?? null });
      }),
    );
    results.push(...scored);
  }
  results.sort((a, b) => b.score - a.score);
  return { results, degraded };
}
```

```ts
import { classifyRoles, ROLE_TARGETS } from "../recs/roles";
import type { PoolCard } from "../recs/score";

export type Suggestion = { oracleId: string; reason: "edhrec" | "ramp" | "draw" | "removal" | "land" };
const MAX = 99;
const LAND_TARGET = 36;

export function suggestDeck(args: {
  commander: PoolCard;
  ownedPool: PoolCard[];
  edhrecTopCards: string[] | null;
}): Suggestion[] {
  const out: Suggestion[] = [];
  const used = new Set<string>();
  const add = (oracleId: string, reason: Suggestion["reason"]) => {
    if (out.length < MAX && !used.has(oracleId)) { used.add(oracleId); out.push({ oracleId, reason }); }
  };

  if (args.edhrecTopCards) {
    const top = new Set(args.edhrecTopCards.map((n) => n.toLowerCase()));
    for (const c of args.ownedPool) {
      if (top.has(c.name.toLowerCase())) add(c.oracleId, "edhrec");
    }
  }

  const counts = { ramp: 0, draw: 0, removal: 0, land: 0 };
  for (const c of args.ownedPool) {
    if (used.has(c.oracleId)) {
      for (const r of classifyRoles(c)) counts[r]++;
    }
  }
  for (const c of args.ownedPool) {
    if (used.has(c.oracleId)) continue;
    for (const role of classifyRoles(c)) {
      const target = role === "land" ? LAND_TARGET : ROLE_TARGETS[role];
      if (counts[role] < target) { add(c.oracleId, role); counts[role]++; break; }
    }
  }
  return out;
}
```

```ts
export async function createDeck(commanderOracleId: string, name: string): Promise<void> {

  // Spec: suggestions are display-only — nothing is committed until the user adds it.

  const inDeck = new Set(
```

```ts
"use server";

import { and, arrayContained, eq, ilike, sql } from "drizzle-orm";
import { db } from "../../../db";
import { cards } from "../../../db/schema";
import { requireUserId } from "../../../lib/auth";

export async function searchCards(
  query: string, colorIdentity: string[], ownedOnly: boolean,
): Promise<{ oracleId: string; name: string; typeLine: string; owned: boolean }[]> {
  const userId = await requireUserId();
  const q = query.trim();
  if (q.length < 2 || q.length > 100) return [];
  // Escape LIKE wildcards so user input can't inject %/_ patterns.
  const escaped = q.replace(/[\\%_]/g, (m) => `\\${m}`);
  const ownedSql = sql<boolean>`exists (select 1 from collection_items ci where ci.oracle_id = ${cards.oracleId} and ci.user_id = ${userId})`;
  // All constraints live in SQL so LIMIT can't starve results when early
  // name matches are off-color or unowned.
  const rows = await db
    .select({
      oracleId: cards.oracleId, name: cards.name, typeLine: cards.typeLine,
      owned: ownedSql,
    })
    .from(cards)
    .where(and(
      ilike(cards.name, `%${escaped}%`),
      arrayContained(cards.colorIdentity, colorIdentity),
      eq(cards.commanderLegal, true),
      ownedOnly ? ownedSql : undefined,
    ))
    .limit(20);
  // Substring (not prefix) match is intentional: "gravetide" must find
  // "Muldrotha, the Gravetide". A scan over the ~35k-row cards table is cheap.
  return rows;
}
```
